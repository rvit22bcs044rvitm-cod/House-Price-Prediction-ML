import streamlit as st
import pandas as pd
import numpy as np
import pickle

# Load assets
model = pickle.load(open('house_model.pkl', 'rb'))
scaler = pickle.load(open('scaler.pkl', 'rb'))

st.title("🏡 House Price Predictor")
st.write("Enter the details below to estimate the house value.")

# match the column order of your 'X' dataframe exactly
# Based on your prompt: Square_Footage, Num_Bedrooms, Num_Bathrooms, Lot_Size, Garage_Size, Neighborhood_Quality, Year_Built
col1, col2 = st.columns(2)

with col1:
    sqft = st.number_input("Square Footage", min_value=100)
    beds = st.number_input("Number of Bedrooms", min_value=1)
    baths = st.number_input("Number of Bathrooms", min_value=1)
    lot = st.number_input("Lot Size (sq ft)", min_value=100)

with col2:
    garage = st.number_input("Garage Size (Cars)", min_value=0)
    quality = st.slider("Neighborhood Quality (1-10)", 1, 10, 5)
    year = st.number_input("Year Built", min_value=1800, max_value=2026)

if st.button("Predict Price"):
    # 1. Arrange inputs into a 2D array
    features = np.array([[sqft, beds, baths, lot, garage, quality, year]])
    
    # 2. Scale the features using the saved scaler
    scaled_features = scaler.transform(features)
    
    # 3. Predict (This gives the LOG price)
    log_prediction = model.predict(scaled_features)
    
    # 4. Inverse Log Transformation (Convert back to actual price)
    final_price = np.expm1(log_prediction)[0]
    
    st.success(f"The estimated house price is: ${final_price:,.2f}")
